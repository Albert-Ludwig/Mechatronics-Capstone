cmake_minimum_required(VERSION 3.22)
project(inventory_pedestrian_visual)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-sim QUIET)
if(gz-sim_FOUND)
  find_package(gz-plugin REQUIRED COMPONENTS register)
  find_package(gz-transport REQUIRED)
  set(gazebo_libraries gz-sim::gz-sim gz-plugin::register gz-transport::core)
else()
  find_package(gz-sim8 REQUIRED)
  find_package(gz-plugin2 REQUIRED COMPONENTS register)
  find_package(gz-transport13 REQUIRED)
  set(gazebo_libraries gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core)
endif()
add_library(inventory_pedestrian_visual SHARED src/smooth_actor.cc)
target_compile_features(inventory_pedestrian_visual PRIVATE cxx_std_17)
target_link_libraries(inventory_pedestrian_visual PRIVATE ${gazebo_libraries})
install(TARGETS inventory_pedestrian_visual LIBRARY DESTINATION lib)
ament_package()
